      let active =

        try {
          const {
            data: {
              user,
            },
            error:
              userError,
          } =
            await supabase.auth
              .getUser();


          if (
            userError ||
            !user
          ) {
            window.location.replace(
              "/login?next=/app-home",
            );

            return;
          }


          const {
            data:
              profile,
            error:
              profileError,
          } =
            await supabase
              .from(
                "profiles",
              )
              .select(
                "name,email",
              )
              .eq(
                "id",
                user.id,
              )
              .maybeSingle();


          if (
            profileError
          ) {
            console.error(
              "Failed to load PWA profile:",
              profileError,

              "string" &&

              0

            typeof profile?.email ===
              "string" &&

              .trim()
              .length >
              0

        ) {
